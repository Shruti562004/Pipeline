pipeline {
    agent any

    environment {
        JAVA_HOME = "C:\\Program Files\\Java\\jdk-11.0.15.1"
        PATH = "${JAVA_HOME}\\bin;${PATH}"
        TOMCAT_HOME = 'D:\\Rays Software\\server\\apache-tomcat-9.0.113'
        TOMCAT_WEBAPPS_PATH = "${TOMCAT_HOME}\\webapps"
        TOMCAT_SERVICE_NAME = 'Tomcat9'
        MAVEN_HOME = "C:\\ProgramData\\Jenkins\\.jenkins\\tools\\hudson.tasks.Maven_MavenInstallation\\Maven"
        DOCKER_PATH = "C:\\Users\\Lenovo\\Desktop\\Project 4 docker"
    }

    stages {

        stage('Cleanup Workspace') {
            steps {
                deleteDir()
            }
        }

        stage('Clone Repository') {
            steps {
                git url: 'https://github.com/Shruti562004/ORSProject-04.git', branch: 'main'
            }
        }

        stage('Build WAR File') {
    steps {
        dir('ORSProject-04') {
            bat '"%MAVEN_HOME%\\bin\\mvn.cmd" clean package -DskipTests'
        }
    }
}

        stage('Copy WAR to Tomcat') {
    steps {
        dir('ORSProject-04') {   // ✅ IMPORTANT FIX
            script {

                def warFile = "target\\ORSProject-04.war"
                def destination = "${TOMCAT_WEBAPPS_PATH}\\ORSProject-04.war"
                def destination_docker = "${DOCKER_PATH}\\ORSProject-04.war"

                bat "rmdir /S /Q \"${TOMCAT_WEBAPPS_PATH}\\ORSProject-04\" || exit 0"
                bat "del /Q \"${TOMCAT_WEBAPPS_PATH}\\ORSProject-04.war\" || exit 0"

                bat "copy /Y \"${warFile}\" \"${destination}\""
                bat "copy /Y \"${warFile}\" \"${destination_docker}\""
            }
        }
    }
}

        stage('Ensure Tomcat Service Exists') {
            steps {
                script {
                    def serviceExists = bat(script: "sc query ${TOMCAT_SERVICE_NAME}", returnStatus: true) == 0

                    if (!serviceExists) {
                        echo "Installing Tomcat service..."
                        bat "cd /d ${TOMCAT_HOME}\\bin && service.bat install"
                    } else {
                        echo "Tomcat already installed."
                    }
                }
            }
        }

        stage('Restart Tomcat') {
            steps {
                script {

                    def isRunning = bat(
                        script: "sc query ${TOMCAT_SERVICE_NAME} | findstr /I \"RUNNING\"",
                        returnStatus: true
                    )

                    if (isRunning == 0) {
                        echo "Stopping Tomcat..."
                        bat "sc stop ${TOMCAT_SERVICE_NAME}"
                        sleep time: 5, unit: 'SECONDS'
                    }

                    echo "Starting Tomcat..."
                    bat "sc start ${TOMCAT_SERVICE_NAME}"
                }
            }
        }

        stage('Verify Deployment') {
            steps {
                bat "powershell -Command \"try {Invoke-WebRequest -Uri 'http://localhost:8080/ORSProject-04' -UseBasicParsing} catch {Write-Host 'Tomcat not reachable'}\""
            }
        }
    }

    post {
        always {
            echo 'Cleaning workspace...'
            deleteDir()
        }
    }
}
